Once loaded, open the browser console. A ready-made graph instance and the raw Module are both attached to window, so you can poke at the API directly:

// already created for you on load:
graph                       // an AudioGraph instance
osc                         // an OscillatorNode* (from graph.addNode)
out                         // an OutputNode*

osc.setFrequency(220);
osc.setSampleRate(44100);
graph.connectNodes(osc, out);

const buf = graph.processBuffer(); // one CHUNKSIZE-length Float array
console.log(buf.length, buf.slice(0, 10));

// or build your own graph from scratch:
const g2 = new Module.AudioGraph();
const o2 = g2.addNode("OscillatorNode");
const gain = g2.addNode("GainNode");
const out2 = g2.addNode("OutputNode");
g2.connectNodes(o2, gain);
g2.connectNodes(gain, out2);
o2.setFrequency(440);
gain.setGainControl(0.5);
g2.processBuffer();
